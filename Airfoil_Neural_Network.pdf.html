108 upper_surface = []
109 lower_surface = []
110
111
112 def mean_camber_line(x):
113
114 if x < p:
115 yc = (m / p**2) * (2 * p * x - x**2)
116
117 else:
118 yc = (m / (1 - p)**2) * (
119 1 - 2 * p + 2 * p * x - x**2
120 )
121
122 return yc
123
124
125 def thickness_distribution(x):
126
127 ty = 5 * t * (
128 0.2969 * math.sqrt(x)
129 - 0.1260 * x
130 - 0.3516 * x**2
131 + 0.2843 * x**3
132 - 0.1015 * x**4
133 )
134
135 return ty
136
137
138 def mean_camber_line_derivative(x):
139
140 if x < p:
141 yc_prime = (2 * m / p**2) * (p - x)
142
143 else:
144 yc_prime = (2 * m / (1 - p)**2) * (p - x)
145
146 return yc_prime
147
148
149 def angle_theta(yc_prime):
150
151 theta = math.atan(yc_prime)
152
153 return theta
154
155
156 def upper_surface_coordinates(x, yt, theta, yc):
157
158 xu = x - yt * math.sin(theta)
159 yu = yc + yt * math.cos(theta)
160
161 upper_surface.append((xu, yu))
162
163 return xu, yu
164
165
166 def lower_surface_coordinates(x, yt, yc, theta):
167
168 xl = x + yt * math.sin(theta)
169 yl = yc - yt * math.cos(theta)
170
171 lower_surface.append((xl, yl))
172
173 return xl, yl

643class ANN(nn.Module):

645 def __init__(self, number_of_neurons):
646
647 super().__init__()

653 # Leaky ReLU activation function
654 def forward(self, x):

656 ReLU_alpha = 0.01

658 x = F.leaky_relu(self.fc1(x), ReLU_alpha)
659 x = F.leaky_relu(self.fc2(x), ReLU_alpha)
660
661 x = self.output(x)
662
663 return x
